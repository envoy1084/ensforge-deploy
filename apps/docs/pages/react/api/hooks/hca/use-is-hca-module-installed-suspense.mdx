# useIsHcaModuleInstalledSuspense

Check a module type/address. Uses the SDK from `EnsforgeProvider`.

## Import

```tsx
import { useIsHcaModuleInstalledSuspense } from "@ensforge/react";
```

## Usage

:::code-group

```tsx [component.tsx]
import { useIsHcaModuleInstalledSuspense } from "@ensforge/react";

function Component() {
  const result = useIsHcaModuleInstalledSuspense({
    hca: "0x1234567890123456789012345678901234567890",
    moduleTypeId: 1n,
    module: "0x5678901234567890123456789012345678901234",
  });

  return (
    <pre>
      {JSON.stringify(
        result.data,
        (_, value) => (typeof value === "bigint" ? value.toString() : value),
        2,
      )}
    </pre>
  );
}
```

## Parameters

```ts
import type { UseEnsSuspenseAtomParameters } from "@ensforge/react";
import type { HcaReadParameters } from "@ensforge/sdk/hca";
```

### hca

`` `0x${string}` ``

HCA address to inspect or operate on.

### moduleTypeId

`bigint`

ERC-7579 module type identifier.

### module

`` `0x${string}` ``

Module contract address.

### additionalContext

`` `0x${string}` | undefined ``

Module-specific inspection context.

### blockNumber

`bigint | undefined`

Block number to read. Cannot be combined with blockTag.

### blockTag

`BlockTag | undefined`

Named block state to read. Cannot be combined with blockNumber.
