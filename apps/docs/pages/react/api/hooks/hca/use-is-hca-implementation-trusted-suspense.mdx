# useIsHcaImplementationTrustedSuspense

Check trusted implementation membership Uses the SDK from `EnsforgeProvider`.

## Import

```tsx
import { useIsHcaImplementationTrustedSuspense } from "@ensforge/react";
```

## Usage

:::code-group

```tsx [component.tsx]
import { useIsHcaImplementationTrustedSuspense } from "@ensforge/react";

function Component(parameters: Parameters<typeof useIsHcaImplementationTrustedSuspense>[0]) {
  const result = useIsHcaImplementationTrustedSuspense(parameters);
  return <p>{result.isWaiting ? "Loading" : "Result available"}</p>;
}
```

## Parameters

```ts
type Input = Parameters<typeof useIsHcaImplementationTrustedSuspense>[0];
```

### blockNumber

`bigint | undefined`

Block number to read. Cannot be combined with blockTag.

### blockTag

`BlockTag | undefined`

Named block state to read. Cannot be combined with blockNumber.

### implementation

`` `0x${string}` ``

Implementation address to inspect, approve or use for derivation.
