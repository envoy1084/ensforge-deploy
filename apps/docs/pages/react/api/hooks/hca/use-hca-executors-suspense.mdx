# useHcaExecutorsSuspense

Inspect executors with pagination Uses the SDK from `EnsforgeProvider`.

## Import

```tsx
import { useHcaExecutorsSuspense } from "@ensforge/react";
```

## Usage

:::code-group

```tsx [component.tsx]
import { useHcaExecutorsSuspense } from "@ensforge/react";

function Component(parameters: Parameters<typeof useHcaExecutorsSuspense>[0]) {
  const result = useHcaExecutorsSuspense(parameters);
  return <p>{result.isWaiting ? "Loading" : "Result available"}</p>;
}
```

## Parameters

```ts
type Input = Parameters<typeof useHcaExecutorsSuspense>[0];
```

### blockNumber

`bigint | undefined`

Block number to read. Cannot be combined with blockTag.

### blockTag

`BlockTag | undefined`

Named block state to read. Cannot be combined with blockNumber.

### hca

`` `0x${string}` ``

HCA address to inspect or operate on.

### cursor

`` `0x${string}` | undefined ``

Starting address for module pagination.

### size

`bigint | undefined`

Maximum number of module entries to return.
