# useHcaFallbackHandlerSuspense

Inspect the handler for a selector. Uses the SDK from `EnsforgeProvider`.

## Import

```tsx
import { useHcaFallbackHandlerSuspense } from "@ensforge/react";
```

## Usage

:::code-group

```tsx [component.tsx]
import { useHcaFallbackHandlerSuspense } from "@ensforge/react";

function Component() {
  const result = useHcaFallbackHandlerSuspense({
    hca: "0x1234567890123456789012345678901234567890",
    selector: "0xa9059cbb",
  });

  return (
    <pre>
      {JSON.stringify(
        result.data,
        (_, value) => (typeof value === "bigint" ? value.toString() : value),
        2,
      )}
    </pre>
  );
}
```

## Parameters

```ts
import type { UseEnsSuspenseAtomParameters } from "@ensforge/react";
import type { HcaReadParameters } from "@ensforge/sdk/hca";
```

### hca

`` `0x${string}` ``

HCA address to inspect or operate on.

### selector

`` `0x${string}` ``

Function selector whose fallback handler is inspected.

### blockNumber

`bigint | undefined`

Block number to read. Cannot be combined with blockTag.

### blockTag

`BlockTag | undefined`

Named block state to read. Cannot be combined with blockNumber.
