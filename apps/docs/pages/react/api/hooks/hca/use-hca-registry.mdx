# useHcaRegistry

Inspect module registry state Uses the SDK from `EnsforgeProvider`.

## Import

```tsx
import { useHcaRegistry } from "@ensforge/react";
```

## Usage

:::code-group

```tsx [component.tsx]
import { useHcaRegistry } from "@ensforge/react";

function Component(parameters: Parameters<typeof useHcaRegistry>[0]) {
  const result = useHcaRegistry(parameters);
  return <p>{result.isWaiting ? "Loading" : "Result available"}</p>;
}
```

## Parameters

```ts
type Input = Parameters<typeof useHcaRegistry>[0];
```

### blockNumber

`bigint | undefined`

Block number to read. Cannot be combined with blockTag.

### blockTag

`BlockTag | undefined`

Named block state to read. Cannot be combined with blockNumber.

### hca

`` `0x${string}` ``

HCA address to inspect or operate on.
