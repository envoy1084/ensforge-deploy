# useVerifyHcaSignatureSuspense

Validate a signature; RPC failures remain errors. Uses the SDK from `EnsforgeProvider`.

## Import

```tsx
import { useVerifyHcaSignatureSuspense } from "@ensforge/react";
```

## Usage

:::code-group

```tsx [component.tsx]
import { useVerifyHcaSignatureSuspense } from "@ensforge/react";
import type { Hex } from "viem";

function Component({ hash, signature }: { hash: Hex; signature: Hex }) {
  const result = useVerifyHcaSignatureSuspense({
    hca: "0x1234567890123456789012345678901234567890",
    hash,
    signature,
  });

  return (
    <pre>
      {JSON.stringify(
        result.data,
        (_, value) => (typeof value === "bigint" ? value.toString() : value),
        2,
      )}
    </pre>
  );
}
```

## Parameters

```ts
import type { UseEnsSuspenseAtomParameters } from "@ensforge/react";
import type { HcaReadParameters } from "@ensforge/sdk/hca";
```

### hca

`` `0x${string}` ``

HCA address to inspect or operate on.

### hash

`` `0x${string}` ``

Digest to inspect or verify.

### signature

`` `0x${string}` ``

Encoded signature to validate.

### sender

`` `0x${string}` | undefined ``

Caller address used during signature verification.

### blockNumber

`bigint | undefined`

Block number to read. Cannot be combined with blockTag.

### blockTag

`BlockTag | undefined`

Named block state to read. Cannot be combined with blockNumber.
