# useIsHcaSessionEnabledSuspense

Check a permission ID on the destination validator. Uses the SDK from `EnsforgeProvider`.

## Import

```tsx
import { useIsHcaSessionEnabledSuspense } from "@ensforge/react";
```

## Usage

:::code-group

```tsx [component.tsx]
import { useIsHcaSessionEnabledSuspense } from "@ensforge/react";
import type { Hex } from "viem";

function Component({ permissionId }: { permissionId: Hex }) {
  const result = useIsHcaSessionEnabledSuspense({
    hca: "0x1234567890123456789012345678901234567890",
    permissionId,
  });

  return (
    <pre>
      {JSON.stringify(
        result.data,
        (_, value) => (typeof value === "bigint" ? value.toString() : value),
        2,
      )}
    </pre>
  );
}
```

## Parameters

```ts
import type { UseEnsSuspenseAtomParameters } from "@ensforge/react";
import type { HcaReadParameters } from "@ensforge/sdk/hca";
```

### hca

`` `0x${string}` ``

HCA address to inspect or operate on.

### permissionId

`` `0x${string}` ``

Permission identifier of the prepared or enabled session.

### blockNumber

`bigint | undefined`

Block number to read. Cannot be combined with blockTag.

### blockTag

`BlockTag | undefined`

Named block state to read. Cannot be combined with blockNumber.
